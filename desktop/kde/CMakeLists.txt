cmake_minimum_required(VERSION 3.16)
project(CompositorDesktopPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
set(KDE_INSTALL_PLUGINDIR "lib/qt6/plugins")
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(KF6KIO REQUIRED)
find_package(KF6CoreAddons REQUIRED)
kcoreaddons_add_plugin(acompositorpreview SOURCES CompositorPreview.cpp INSTALL_NAMESPACE "kf6/thumbcreator")
set_target_properties(acompositorpreview PROPERTIES PREFIX "")
target_link_libraries(acompositorpreview PRIVATE KF6::KIOGui KF6::CoreAddons Qt6::Gui)
enable_testing()
add_executable(test_compositor_preview test_preview.cpp)
target_link_libraries(test_compositor_preview PRIVATE KF6::KIOGui KF6::CoreAddons Qt6::Gui)
add_test(NAME compositor_preview COMMAND test_compositor_preview $<TARGET_FILE:acompositorpreview>)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_test(NAME compositor_preview_dispatch COMMAND ${DBUS_RUN_SESSION} $<TARGET_FILE:test_compositor_preview> $<TARGET_FILE:acompositorpreview> --dispatch)
